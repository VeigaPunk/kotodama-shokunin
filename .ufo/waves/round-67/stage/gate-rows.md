else:
    q, words = c[6], c[6].strip('"').split()
    if len(words) > 25:
        w = words[:25]
        while w and w[-1].lower() in {"a","an","the","of","and","or","to","in","on",
                                      "for","with","at","by","as","is","are","was","were"}:
            w.pop()

    print(f"- {sid} | {url} | 2026-10-24 | {q}")

done
