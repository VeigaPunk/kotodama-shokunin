<script>
let currentSlide = 0;
const slides = document.querySelectorAll('.slide');
let startX = 0;

function openModal(index) {
    document.getElementById('myModal').style.display = "block";
    showSlide(index);
}

function closeModal() {
    document.getElementById('myModal').style.display = "none";
}

function changeSlide(n) {
    showSlide(currentSlide + n);
}

function showSlide(n) {
    const total = slides.length;
    if (n >= total) currentSlide = 0;
    else if (n < 0) currentSlide = total - 1;
    else currentSlide = n;

    slides.forEach((slide, i) => {
        slide.style.display = i === currentSlide ? "block" : "none";
    });
}

document.addEventListener("DOMContentLoaded", () => {
    showSlide(0);
});

// Swipe support
const modalContent = document.getElementById('modalContent');
modalContent.addEventListener('touchstart', e => { startX = e.touches[0].clientX; });
modalContent.addEventListener('touchend', e => {
    const endX = e.changedTouches[0].clientX;
    if (startX - endX > 50) changeSlide(1);
    if (startX - endX < -50) changeSlide(-1);
});
</script>

            <div class="ins_notes">
                <span class="note" id="note-1"><a href="#ref-1">[1]</a> This is one of a number of imperial period dedications to Zeus Hypsistos (“Zeus The All High”) deriving from the former site of the Athenian Assembly on the Pnyx. For other such dedications see <a class="highlight" href="https://www.atticinscriptions.com/inscription/AIUK45/13"><i>AIUK</i> 4.5 no. 13</a>, with notes.  Many of them take the form of anatomical votives. In this case, a single breast is carved in relief beneath the inscription. Votive breasts are a common form of dedication, and several examples are known from this sanctuary (see <a class="highlight" href="https://www.atticinscriptions.com/inscription/AIUK45/13"><i>AIUK</i> 4.5 no. 13</a>). They must have some connection with female health, although we need not necessarily assume that the dedicant was suffering from a disease directly afflicting the breast. A connection with breast-feeding, or fertility more generally, is also possible.</span>
