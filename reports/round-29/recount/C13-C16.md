## C14 — RL-4: the eight records reach the live packet (+8 rows)

```sh

python3 -B - <<'PY'
from pathlib import Path
import hashlib, runpy
load = runpy.run_path('tools/gates/naming-collision-gate.py')['load_yaml']
required = {f'OR-0{i}' for i in range(1, 4)} | {f'LI-0{i}' for i in range(1, 4)} | {f'EN-0{i}' for i in range(1, 3)}
paths = {
    'r27_frozen': Path('reports/round-27/pinsweep/frozen/r27-post/inputs/examples/culture-d4/packet.d4.yml'),
    'r28_live': Path('examples/culture-d4/packet.d4.yml'),
}
for label, path in paths.items():
    raw = path.read_bytes()
    packet = load(raw.decode('utf-8'))
    rows = [row for value in packet.values() if isinstance(value, list) for row in value if isinstance(row, dict) and row.get('id')]
    ids = {row['id'] for row in rows}
    present = sorted(ids & required)

PY
```
