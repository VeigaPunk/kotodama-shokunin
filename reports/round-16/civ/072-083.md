```bash
python3 - <<'EOF'
import re, os
BASE='corpus/knowledge'; CIV='corpus/knowledge/civilizations'
expect={'persia-sasanian.md':16,'polynesia-navigation.md':15,'rapa-nui.md':15,
        'rome.md':15,'ryukyu.md':11,'scythian-saka.md':14,'siam-thai.md':15}
def expand_chunk(ch):
    m=re.search(r'\{([^}]*)\}',ch)
    if not m: return [ch]
    pre=ch[:m.start()]; alts=m.group(1); suf=ch[m.end():]
    d=re.search(r'((?:[A-Za-z0-9_.-]+/)+)$',pre)
    base=d.group(1) if d else ''
    return [base+a+suf for a in alts.split(',')]
def pathify(t):
    m=re.search(r'((?:\.\./|corpus/knowledge/)?(?:linguistics|myth-symbol|culture-material|synthesis|verification)/[A-Za-z0-9_./-]*\.md)$',t)
    return m.group(1) if m else None
def resolve(t,anchor):
    if '/' not in t:
        if not anchor: return None
        t=anchor+t
    if t.startswith('../'): cands=[os.path.normpath(os.path.join(CIV,t))]
    elif t.startswith('corpus/'): cands=[t]
    else: cands=[os.path.join(BASE,t)]
    for c in cands:
        if os.path.isfile(c): return c
    return None
files=sorted(os.listdir(CIV))[71:83]
assert len(files)==12 and files[0]=='olmec.md' and files[-1]=='siam-thai.md'
allok=True
for f in files:
    text=open(f'{CIV}/{f}',encoding='utf-8').read()
    hdr=re.split(r'^## ',text,maxsplit=1,flags=re.M)[0]
    fields={k:sum(1 for l in hdr.splitlines() if re.match(rf'^- \*?\*?{k}\*?\*?:',l))
            for k in ('topic','sources','couples','source','verified-at','recheck')}
    assert all(v==1 for v in fields.values()), (f,fields)
    raw=next(l for l in hdr.splitlines() if l.startswith('- couples:'))[len('- couples:'):]
    chunks=[c for ch in raw.split() for c in expand_chunk(ch)]
    resolved=[];anchor=None
    for ch in chunks:
        t=ch.strip('`,;: \t')
        if not t: continue
        if t.endswith('/'):
            a=re.search(r'((?:[A-Za-z0-9_.-]+/)+)$',t)
            if a: anchor=a.group(1)
            continue
        p=pathify(t) or (t if re.fullmatch(r'[A-Za-z0-9_.-]+\.md',t) else None)
        if not p: continue
        r=resolve(p,anchor)
        if r:
            resolved.append(r)
            if '/' in p:
                d=os.path.dirname(p)
                anchor=('../'+os.path.relpath(d,'corpus/knowledge/civilizations')+'/') if p.startswith('../') else (d+'/')
    uniq=sorted(set(resolved))
    urls=len({u.rstrip('.,;') for u in re.findall(r'https?://[^\s)>\"`]+',text)})
    line=f"{f:26} fields=6/6 couples={len(uniq)} distinct_urls={urls}"
    if f in expect:
        m=len(uniq)==expect[f]; allok&=m
        line+=f" v05={expect[f]} {'MATCH' if m else 'MISMATCH'}"
    print(line)
print("ALL_MATCH" if allok else "SOME_MISMATCH")
EOF
```
